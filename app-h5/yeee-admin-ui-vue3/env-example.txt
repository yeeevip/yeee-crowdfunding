# 环境变量配置示例文件
# 请复制此文件内容创建对应的 .env 文件

# === .env ===
# 基础环境变量
VITE_APP_TITLE=一页一后台管理系统
VITE_APP_VERSION=v3.0.0

# API 配置
VITE_API_BASE_URL=/api
VITE_API_TIMEOUT=10000

# 其他配置
VITE_ROUTER_HISTORY=hash
VITE_PUBLIC_PATH=/

# === .env.development ===
# 开发环境配置
NODE_ENV=development

# 开发环境 API 配置
VITE_API_BASE_URL=/api
VITE_API_TARGET=http://localhost:8801

# 开发环境特定配置
VITE_OPEN_DEVTOOLS=true
VITE_MOCK_ENABLED=false

# 调试配置
VITE_DEBUG=true
VITE_LOG_LEVEL=debug

# === .env.production ===
# 生产环境配置
NODE_ENV=production

# 生产环境 API 配置
VITE_API_BASE_URL=/api
VITE_API_TARGET=https://your-production-api.com

# 生产环境特定配置
VITE_OPEN_DEVTOOLS=false
VITE_MOCK_ENABLED=false

# 性能配置
VITE_BUILD_GZIP=true
VITE_BUILD_ANALYZE=false

# 调试配置
VITE_DEBUG=false
VITE_LOG_LEVEL=error
